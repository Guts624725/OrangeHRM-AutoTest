提取 token 的标准模板:
    import html
    import re

    token_match = re.search(r':token="([^"]+)"', response.text)
    if not token_match:
        raise ValueError("Token 提取失败")

    token = token_match.group(1)
    token = html.unescape(token)      # 第1步：解码 HTML 实体
    token = token.replace('"', '')    # 第2步：去除残留引号
    token = token.strip()             # 第3步：去除首尾空格
